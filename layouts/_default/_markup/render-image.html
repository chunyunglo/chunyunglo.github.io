{{- /* Page-bundle images become responsive WebP (no EXIF) with lazy loading; others pass through. */ -}} {{- $img := .Page.Resources.Get .Destination -}} {{- if and $img (in (slice "jpeg" "png" "webp") $img.MediaType.SubType) -}} {{- $w := cond (gt $img.Width 960) 960 $img.Width -}} {{- $a := $img.Resize (printf "%dx webp" $w) -}} {{- $srcset := printf "%s %dw" $a.RelPermalink $a.Width -}} {{- if gt $img.Width 1200 }}{{ $b := $img.Resize "1600x webp" }}{{ $srcset = printf "%s, %s %dw" $srcset $b.RelPermalink $b.Width }}{{ end -}} {{- $sizes := "(max-width: 960px) 100vw, 720px" -}} {{- if not .IsBlock }}{{ $t := $img.Resize "480x webp" }}{{ $srcset = printf "%s %dw, %s" $t.RelPermalink $t.Width $srcset }}{{ $sizes = "(max-width: 600px) 50vw, 240px" }}{{ end -}} {{- $tag := printf `` $a.RelPermalink $srcset $sizes $a.Width $a.Height .Text -}} {{- if .IsBlock -}} {{- else -}} {{ $tag | safeHTML }} {{- end -}} {{- else -}} {{- end -}}
