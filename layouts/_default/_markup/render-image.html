{{- /* Page-bundle images become responsive WebP with lazy loading; others pass through. */ -}} {{- $img := .Page.Resources.Get .Destination -}} {{- if and $img (in (slice "jpeg" "png" "webp") $img.MediaType.SubType) -}} {{- $w := cond (gt $img.Width 960) 960 $img.Width -}} {{- $a := $img.Resize (printf "%dx webp" $w) -}} {{- $b := $a -}}{{- if gt $img.Width 1920 }}{{ $b = $img.Resize "1920x webp" }}{{ end -}} {{- else -}} {{- end -}}
